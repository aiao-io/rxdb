import '@analogjs/vitest-angular/setup-snapshots';
import '@angular/compiler';
import { setupTestBed } from '@analogjs/vitest-angular/setup-testbed';
import { getTestBed } from '@angular/core/testing';
import { afterAll, beforeEach, expect } from 'vitest';

const marker = Symbol.for('testbed-setup');
const trackingKey = Symbol.for('review-testbed-identities');
type Tracking = { ids: WeakMap<object, number>; next: number };
const tracking = (Reflect.get(globalThis, trackingKey) as Tracking | undefined) ?? {
  ids: new WeakMap<object, number>(),
  next: 1
};
Reflect.set(globalThis, trackingKey, tracking);

function trace(stage: string): void {
  const bed = getTestBed();
  if (!tracking.ids.has(bed)) tracking.ids.set(bed, tracking.next++);
  console.log('REVIEW_TESTBED ' + JSON.stringify({
    stage,
    testPath: expect.getState().testPath,
    identity: tracking.ids.get(bed),
    marker: Reflect.get(globalThis, marker) === true,
    platform: Boolean(bed.platform),
    ngModule: Boolean(bed.ngModule),
    control: process.env['REVIEW_TESTBED_REPAIR'] === 'true'
  }));
}

trace('before-analog');
if (process.env['REVIEW_TESTBED_REPAIR'] === 'true' && !getTestBed().platform) {
  Reflect.deleteProperty(globalThis, marker);
}
setupTestBed();
trace('after-analog');
beforeEach(() => trace('beforeEach'));
afterAll(() => trace('afterAll'));

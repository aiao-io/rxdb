 176  * @remarks
 177  * 每次调用生成独立的注入 key，因此多次调用得到的是**互相隔离**的 provider/injector
 178  * （与 React 的 `makeRxDBProvider` 每次新建 context 对齐）。key 若提到模块顶层，
 179  * 两套注入器会读写同一个 Symbol，后 provide 的数据库覆盖前者，而泛型 `T` 只是虚假的
 180  * 编译期断言 —— 拿到的可能是另一套注入的数据库。
 181  */
 182 function makeRxDBDependencyInjector<T extends RxDB>(): RxDBDependencyInjection<T> {
 183   const { provideRxDB, injectRxDB, injectRxDBRef } = createInjection<T>();
 184   return { provideRxDB, injectRxDB, injectRxDBRef };
 185 }
 186 
 187 /**
 188  * 包级默认注入器。
 189  *
 190  * @remarks
 191  * 绝大多数应用只有一个数据库，直接用这几个导出即可 ——
 192  * 它们共享同一个模块级注入 key，`provideRxDB` 与 `injectRxDB*` 天然配对。
 193  *
 194  * - `provideRxDB(db)`：在当前组件的 setup 中提供数据库，作用域是该组件的整棵子树。
 195  * - `injectRxDB()`：取当前已就绪的实例，无 provider 或数据库未就绪时返回 `undefined`。
 196  * - `injectRxDBRef()`：取响应式引用，用于等待异步就绪；无 provider 时返回 `undefined`
 197  *   （dev 下 Vue 会额外发出 injection 未找到的警告）。
 198  *
 199  * 需要在同一棵组件树里隔离多个数据库时，改用 {@link makeRxDBDependencyInjector}。
 200  * 需要「没有就抛错」而非返回 `undefined` 的语义时，用 {@link useRxDB} / {@link useRxDBRef}。
 201  */
 202 const { injectRxDB, injectRxDBRef, injectSlot, injectSlotQuietly, provideRxDB } = createInjection<RxDB>();
 203 
 204 /**
 205  * 获取已提供的 RxDB 响应式引用。
 206  *
 207  * @returns 当前 provider 中的只读引用，允许数据库异步就绪。
 208  * @throws 未找到 RxDB provider 时抛出错误。
 209  */
 210 const useRxDBRef = (): RxDBRef<RxDB> => {
 211   const databaseRef = injectRxDBRef();
 212   if (!databaseRef) {
 213     throw new Error(RXDB_NOT_FOUND_MESSAGE);
 214   }
 215   return databaseRef;
 216 };
 217 
 218 /**
 219  * 获取当前已就绪的 RxDB 实例。
 220  *
 221  * @returns 当前 RxDB 实例。
 222  * @throws 没有 provider、数据库尚未就绪时各抛一条不同的文案；异步 source 创建失败时
 223  *   原样抛出创建异常（与 Angular / React 同语义）。
 224  */
 225 const useRxDB = (): RxDB => {
 226   const slot = injectSlot();
 227   if (!slot) {
 228     throw new Error(RXDB_NOT_FOUND_MESSAGE);
 229   }
 230   if (slot.failure.value !== undefined) {
 231     throw slot.failure.value;
 232   }
 233   if (!slot.db.value) {
 234     throw new Error(RXDB_NOT_READY_MESSAGE);
 235   }
 236   return slot.db.value;
 237 };
 238 
 239 /**
 240  * 获取当前已就绪的 RxDB 实例，取不到时返回 `undefined`。
 241  *
 242  * @returns 已就绪的实例；没有 provider 或数据库尚未就绪时为 `undefined`
 243  *
 244  * @remarks
 245  * 与 {@link useRxDB} 的区别只有「没有时怎么办」：这个不抛错，用于异步 source 下渲染 loading 态。
 246  * 语义与 {@link injectRxDB} 相同，区别只有缺 provider 时**不发** Vue 的 injection 告警 ——
 247  * 可选读取本就允许没有 provider。名字与 Angular / React 的 `useRxDBOptional` 对齐。
 248  */
 249 const useRxDBOptional = (): RxDB | undefined => injectSlotQuietly()?.db.value;
 250 
 251 export { injectRxDB, injectRxDBRef, makeRxDBDependencyInjector, provideRxDB, useRxDB, useRxDBOptional, useRxDBRef };

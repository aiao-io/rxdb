 232   #renderMarkers(view: ReadyView, markers: readonly ReplayCommitMarker[]): HTMLUListElement {
 233     const list = element('ul', { className: 'rxdb-replayer__commits' }, { 'aria-label': 'Commits' });

 236       const button = element('button', {
 237         type: 'button',

 241       button.addEventListener('click', () => this.#restore(view, marker, offset));
 242       const item = element('li');

 249   async #restore(view: ReadyView, marker: ReplayCommitMarker, offset: number): Promise<void> {
 250     const generation = this.#generation;
 251     this.pause();
 252     this.seek(offset);
 253     view.status.textContent = 'Restoring…';
 254     let result: ReplayerCommitRestoreEvent['result'];
 255     try {
 256       result = await this.#options.replay.restoreToCommit(marker.commitId);
 257     } catch (error) {
 258       result = { ok: false, reason: 'error', error: toError(error) };
 259     }
 260     if (generation !== this.#generation) return;
 261     view.status.textContent = this.#describe(result);
 262     this.#options.onCommitRestore?.({ marker, result });
 263   }
 264 
 265   #describe(result: ReplayerCommitRestoreEvent['result']): string {
 266     if (result.ok) return `Restored ${result.restoredCount} changes`;
 267     return result.reason === 'error' ? result.error.message : replayRestoreHint(result.reason);
 268   }

 270   #finish(view: ReadyView): void {
 271     this.#stopTicking();
 272     this.#playing = false;
 273     this.#setPosition(view, view.total);
 274     this.#renderToggle(view);
 275   }

 277   #tick(): void {
 278     const view = this.#view;
 279     if (view === null || !this.#playing) return;
 280     const time = Math.min(view.replayer.getCurrentTime(), view.total);
 281     if (time !== this.#position) this.#setPosition(view, time);
 282     this.#frame = requestAnimationFrame(() => this.#tick());
 283   }
 284 
 285   #stopTicking(): void {
 286     if (this.#frame !== null) cancelAnimationFrame(this.#frame);
 287     this.#frame = null;
 288   }

 290   #setPosition(view: ReadyView, time: number): void {
 291     this.#position = time;
 292     view.timeline.value = String(time);
 293     view.timeline.setAttribute('aria-valuetext', mmss(time));
 294     this.#options.onTimeChange?.(time);

 318 /**
 319  * 在 `host` 里渲染一个会话的回放器：rrweb 回放、播放 / 暂停、时间轴、commit 标记（点了恢复工作树）与恢复结果。
 320  *
 321  * @remarks
 322  * DOM 与四态（`loading` / `empty` / `error` / `ready`）见 `specs/005-us-909-session-replay/contracts/replayer-component.md` §2；
 323  * 三个框架的 `Replayer` 组件都只是把宿主元素交给它。`rrweb` 在加载时才动态导入。
 324  *
 325  * @param host - 回放器挂载的元素，原有子节点会被替换
 326  * @param options - 会话来源与回调
 327  * @returns 命令句柄；卸载时调 `destroy()`
 328  *
 329  * @example
 330  * ```ts
 331  * const handle = mountReplayer(element, { replay: rxdb.replay, sessionId });
 332  * handle.seek(5_000);
 333  * handle.destroy();
 334  * ```
 335  */
 336 export const mountReplayer = (host: HTMLElement, options: ReplayerOptions): ReplayerHandle => {
 337   const view = new ReplayerView(host, options);
 338   return {
 339     update: next => view.update(next),
 340     play: () => view.play(),
 341     pause: () => view.pause(),
 342     seek: timeMs => view.seek(timeMs),
 343     destroy: () => view.destroy()
 344   };
 345 };

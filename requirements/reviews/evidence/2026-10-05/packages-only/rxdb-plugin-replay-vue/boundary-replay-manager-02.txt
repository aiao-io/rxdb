 133       this.#starting = null;

 158   async listCommitMarkers(sessionId: string): Promise<ReplayCommitMarker[]> {
 159     const entries = await this.#store.readCustomEntries(sessionId);
 160     return entries.flatMap(({ seq, event }) => {
 161       const marker = parseReplayMarker(event);
 162       if (marker?.tag !== REPLAY_MARKER_TAGS.commit) return [];
 163       return [{ seq, timestamp: event.timestamp, ...marker.payload }];
 164     });

 167   async exportSession(sessionId: string): Promise<ReplaySessionExport> {
 168     const session = await this.#store.getSession(sessionId);
 169     if (session === null) {
 170       throw new RxDBReplayError('session_not_found', `session "${sessionId}" does not exist`);

 173     return { format: 'aiao-rxdb-replay-session', version: 1, session: toInfo(session), events };

 191   /**

 203   async #startSession(): Promise<string> {
 204     const record = await loadRrwebRecord();
 205     const sessionId = await this.#store.createSession(new Date());
 206     try {
 207       this.#begin(record, sessionId, 0);
 208     } catch (error) {
 209       await this.#store.markStopped(sessionId);
 210       throw error;
 211     }
 212     return sessionId;
 213   }

 215   /** 起录制器、订阅 commit、转 `recording`。`record()` 没起来时同步抛，状态不动。 */
 216   #begin(record: ReplayRecordFn, sessionId: string, nextSeq: number): void {
 217     const recorder: ReplayRecorder = new ReplayRecorder({
 218       sink: this.#store,
 219       record,
 220       sessionId,
 221       nextSeq,
 222       flush: this.#options.flush,
 223       recordOptions: this.#options.record,
 224       onEnd: end => this.#onEnd(recorder, end)
 225     });
 226     recorder.start();
 227     this.#recorder = recorder;
 228     if (hasWorkingTree(this.#rxdb)) {
 229       this.#commits = this.#rxdb.workingTree.commits$.subscribe(event => recorder.addCommitMarker(event));
 230     }
 231     this.#state.next({ kind: 'recording', sessionId });
 232   }

 251   /** 录制器提前结束（截断或冲刷失败）。旧录制器迟到的结局不算数。 */
 252   #onEnd(recorder: ReplayRecorder, end: ReplayRecorderEnd): void {
 253     if (this.#recorder !== recorder) return;
 254     this.#recorder = null;
 255     this.#unsubscribeCommits();
 256     const sessionId = recorder.sessionId;
 257     if (end.kind === 'error') {
 258       this.#state.next({ kind: 'error', sessionId, error: end.error });
 259       return;
 260     }
 261     this.#storage?.removeItem(this.#stashKey);
 262     this.#state.next({ kind: 'truncated', sessionId, code: end.code });
 263   }

 265   async #resumeOrReport(stash: ReplayStash): Promise<void> {
 266     try {
 267       await this.#resume(stash);
 268     } catch (error) {
 269       const cause = error instanceof Error ? error : new Error(String(error));
 270       this.#state.next({ kind: 'error', sessionId: stash.sessionId, error: cause });
 271     }

 274   /**
 275    * 刷新续录（data-model §4）：补写暂存里还没落库的事件，在同一个会话上重新 `record()`。
 276    *
 277    * @remarks
 278    * 会话已不是 `recording`（刷新前已截断，或在别处被停）就不续，保持 `idle`。
 279    * 补写超限时会话就此截断，同样不再起录制器。
 280    */
 281   async #resume(stash: ReplayStash): Promise<void> {
 282     const session = await this.#store.getSession(stash.sessionId);
 283     if (session?.status !== 'recording' || this.#released) return;
 284     const record = await loadRrwebRecord();
 285     const entries = this.#stashEntries(stash, session.nextSeq);
 286     let written = session.nextSeq;
 287     if (entries.length > 0) {
 288       const result = await this.#store.appendBatch(stash.sessionId, entries);
 289       if (result.kind === 'truncated') {
 290         this.#state.next({ kind: 'truncated', sessionId: stash.sessionId, code: result.code });
 291         return;
 292       }
 293       written = result.nextSeq;
 294     }
 295     this.#begin(record, stash.sessionId, Math.max(stash.nextSeq, written));
 296   }

 298   /** 完整暂存：补写未落库的那段；最小暂存：在会话行的 `nextSeq` 上记一个 `gap` 标记。 */
 299   #stashEntries(stash: ReplayStash, sessionNextSeq: number): ReplayEventEntry[] {
 300     if (stash.gap !== true) return unwrittenStashEvents(stash, sessionNextSeq);
 301     const gap = createReplayMarkerEvent(REPLAY_MARKER_TAGS.gap, { reason: 'stash_unavailable' }, Date.now());
 302     return [{ seq: sessionNextSeq, event: gap }];
 303   }

 305   /** `pagehide` 写暂存，`pageshow(persisted)`（从 bfcache 回来，页面没真卸载）删暂存。 */
 306   #watchPage(storage: Storage): () => void {
 307     const onPageHide = (): void => {
 308       const recorder = this.#recorder;
 309       if (recorder === null) return;
 310       writeReplayStash(storage, this.#stashKey, {
 311         sessionId: recorder.sessionId,
 312         nextSeq: recorder.nextSeq,
 313         events: recorder.pending()
 314       });
 315     };
 316     const onPageShow = (event: Event): void => {
 317       if ((event as PageTransitionEvent).persisted) storage.removeItem(this.#stashKey);
 318     };
 319     const page = globalThis as Required<Pick<PageGlobal, 'addEventListener' | 'removeEventListener'>>;
 320     page.addEventListener('pagehide', onPageHide);
 321     page.addEventListener('pageshow', onPageShow);
 322     return () => {
 323       page.removeEventListener('pagehide', onPageHide);
 324       page.removeEventListener('pageshow', onPageShow);
 325     };
 326   }

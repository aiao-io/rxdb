 113  * 三端 `Replayer` 组件共用的 parity 用例（`specs/005-us-909-session-replay/contracts/replayer-component.md` §4）。
 114  *
 115  * @remarks
 116  * 组件测试里 `mountReplayer` 打桩成 {@link MountReplayerSpy}，对每条用例 `spy.reset()` 后 `await testCase.run(driver, spy)`。
 117  * 同一份用例在三端跑：任何一端漏转发、改名、多调一次 `update()` 都会抛错。
 118  *
 119  * @example
 120  * ```ts
 121  * for (const testCase of replayerParityCases) {
 122  *   it(testCase.name, async () => {
 123  *     spy.reset();
 124  *     await testCase.run(driver, spy);
 125  *   });
 126  * }
 127  * ```
 128  */
 129 export const replayerParityCases: readonly ReplayerParityCase[] = [
 130   {
 131     name: '1. 挂载 → mountReplayer 一次，宿主在组件内，输入原样传入',
 132     async run(driver, spy) {
 133       const inputs = createInputs();
 134       await driver.mount(inputs, noOutputs);
 135       const { host, options } = onlyMount(spy);
 136       if (!driver.container().contains(host)) fail('host must be inside the component', host.outerHTML);
 137       if (options.replay !== inputs.replay) fail('replay must be passed through as is', options.replay);
 138       assertSame(
 139         { sessionId: options.sessionId, initialTime: options.initialTime },
 140         { sessionId: inputs.sessionId, initialTime: inputs.initialTime },
 141         'sessionId / initialTime must be passed through'
 142       );
 143       assertSame(spy.calls, [], 'mounting must not call the handle');
 144     }
 145   },
 146   {
 147     name: '2. 改 sessionId → update({ sessionId }) 一次，不重新挂载',
 148     async run(driver, spy) {
 149       await driver.mount(createInputs(), noOutputs);
 150       await driver.setInputs({ sessionId: 's2' });
 151       onlyMount(spy);
 152       assertSame(spy.calls, [{ method: 'update', args: [{ sessionId: 's2' }] }], 'changing sessionId');
 153     }
 154   },
 155   {
 156     name: '3. 改 initialTime → update({ initialTime })',
 157     async run(driver, spy) {
 158       await driver.mount(createInputs(), noOutputs);
 159       await driver.setInputs({ initialTime: 2_000 });
 160       onlyMount(spy);
 161       assertSame(spy.calls, [{ method: 'update', args: [{ initialTime: 2_000 }] }], 'changing initialTime');
 162     }
 163   },
 164   {
 165     name: '4. onTimeChange(1234) → 时刻输出收到 1234',
 166     async run(driver, spy) {
 167       const received: number[] = [];
 168       await driver.mount(createInputs(), { ...noOutputs, timeChange: timeMs => received.push(timeMs) });
 169       onlyMount(spy).options.onTimeChange?.(1_234);
 170       assertSame(received, [1_234], 'time change output');
 171     }
 172   },
 173   {
 174     name: '5. onCommitRestore(e) → 恢复输出收到同一个 e',
 175     async run(driver, spy) {
 176       const received: ReplayerCommitRestoreEvent[] = [];
 177       await driver.mount(createInputs(), { ...noOutputs, commitRestore: event => received.push(event) });
 178       const event: ReplayerCommitRestoreEvent = {
 179         marker: { seq: 1, timestamp: 1, commitId: 'c1', branchId: 'main' },
 180         result: { ok: false, reason: 'dirty_working_tree' }
 181       };
 182       onlyMount(spy).options.onCommitRestore?.(event);
 183       if (received.length !== 1 || received[0] !== event)
 184         fail('commit restore output must get the same event', received);
 185     }
 186   },
 187   {
 188     name: '6. play / pause / seek(500) → 句柄同名方法各一次',
 189     async run(driver, spy) {
 190       await driver.mount(createInputs(), noOutputs);
 191       driver.play();
 192       driver.pause();
 193       driver.seek(500);
 194       assertSame(
 195         spy.calls,
 196         [
 197           { method: 'play', args: [] },
 198           { method: 'pause', args: [] },
 199           { method: 'seek', args: [500] }
 200         ],
 201         'commands'
 202       );
 203     }
 204   },
 205   {
 206     name: '7. 卸载 → destroy() 一次',
 207     async run(driver, spy) {
 208       await driver.mount(createInputs(), noOutputs);
 209       await driver.unmount();
 210       assertSame(spy.argsOf('destroy'), [[]], 'unmounting');
 211     }
 212   }
 213 ];

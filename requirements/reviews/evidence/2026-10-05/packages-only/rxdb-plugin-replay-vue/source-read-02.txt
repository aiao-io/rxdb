  19 type ReplayerWrapper = VueWrapper<InstanceType<typeof Replayer>>;

  21 const createDriver = (): ReplayerParityDriver => {
  22   let wrapper: ReplayerWrapper | undefined;
  23   const mounted = (): ReplayerWrapper => {
  24     if (!wrapper) throw new Error('not mounted');
  25     return wrapper;
  26   };
  27   // expose() 的成员不进组件实例类型，模板 ref 拿到的就是这组命令
  28   const commands = (): ReplayerRef => mounted().vm as unknown as ReplayerRef;
  29 
  30   return {
  31     async mount(inputs, outputs) {
  32       wrapper = mount(Replayer, {
  33         props: { ...inputs, onTimeChange: outputs.timeChange, onCommitRestore: outputs.commitRestore }
  34       });
  35       await nextTick();
  36     },
  37     async setInputs(inputs) {
  38       await mounted().setProps(inputs);
  39     },
  40     play: () => commands().play(),
  41     pause: () => commands().pause(),
  42     seek: timeMs => commands().seek(timeMs),
  43     async unmount() {
  44       mounted().unmount();
  45       await nextTick();
  46     },
  47     container: () => mounted().element
  48   };
  49 };

   1 # @aiao/rxdb-plugin-replay-vue
   2 
   3 [`@aiao/rxdb-plugin-replay`](../rxdb-plugin-replay) 的 Vue 集成层。提供 `Replayer` 组件，把核心包的 `mountReplayer` 包成 props / emits / `expose` 命令。
   4 
   5 ## 安装
   6 
   7 ```bash
   8 pnpm add @aiao/rxdb @aiao/rxdb-plugin-replay @aiao/rxdb-plugin-replay-vue vue rxjs
   9 ```
  10 
  11 ## 用法
  12 
  13 ```vue
  14 <script lang="ts" setup>
  15 import type { ReplayManager } from '@aiao/rxdb-plugin-replay';
  16 import {
  17   Replayer,
  18   replayRestoreHint,
  19   type ReplayerCommitRestoreEvent,
  20   type ReplayerRef
  21 } from '@aiao/rxdb-plugin-replay-vue';
  22 import { ref, useTemplateRef } from 'vue';
  23 
  24 defineProps<{ replay: ReplayManager; sessionId: string }>();
  25 
  26 const replayer = useTemplateRef<ReplayerRef>('replayer');
  27 const time = ref(0);
  28 const hint = ref<string | null>(null);
  29 
  30 function onRestore({ result }: ReplayerCommitRestoreEvent): void {
  31   if (result.ok) hint.value = null;
  32   else hint.value = result.reason === 'error' ? result.error.message : replayRestoreHint(result.reason);
  33 }
  34 </script>
  35 
  36 <template>
  37   <Replayer
  38     :initial-time="0"
  39     :replay="replay"
  40     :session-id="sessionId"
  41     @commit-restore="onRestore"
  42     @time-change="time = $event"
  43     ref="replayer"
  44   />
  45   <button @click="replayer?.play()" type="button">播放</button>
  46   <button @click="replayer?.pause()" type="button">暂停</button>
  47   <p>{{ time }} ms</p>
  48   <p v-if="hint" role="alert">{{ hint }}</p>
  49 </template>
  50 ```
  51 
  52 `replay` 传 `db.replay`。录制插件要先装进连接纪元（`db.use(rxDBPluginReplay, …)` 之后 `await db.connect()`），否则组件加载会话时拿到 `not_installed`。
  53 
  54 | 成员                              | 说明                                                                                                                |
  55 | --------------------------------- | ------------------------------------------------------------------------------------------------------------------- |
  56 | `replay`（必填）                  | `ReplayManager`；换成别的门面会重新加载                                                                             |
  57 | `sessionId`（必填）               | 要回放的会话                                                                                                        |
  58 | `initialTime`                     | 加载完成后落到的时刻（相对会话起点的 ms，默认 0）；之后再改等价于 `seek()`                                          |
  59 | `time-change`                     | 播放 / 拖动时间轴时发出当前时刻（ms）                                                                               |
  60 | `commit-restore`                  | 点 commit 标记恢复工作树后的结果，被拒绝时用 `replayRestoreHint(reason)` 取提示；`reason` 为 `'error'` 时看 `error` |
  61 | `play()` / `pause()` / `seek(ms)` | 经模板 `ref`（`ReplayerRef`）调用；加载完成前调用是空操作（`seek` 会记下目标时刻）                                  |
  62 
  63 组件卸载时自动释放回放器。rrweb 回放端在组件首次加载会话时才按需 `import('rrweb')`。
  64 
  65 三框架同功能对称：Angular 用 `ao-replayer`（[`@aiao/rxdb-plugin-replay-angular`](../rxdb-plugin-replay-angular)），React 用 `Replayer`（[`@aiao/rxdb-plugin-replay-react`](../rxdb-plugin-replay-react)）。

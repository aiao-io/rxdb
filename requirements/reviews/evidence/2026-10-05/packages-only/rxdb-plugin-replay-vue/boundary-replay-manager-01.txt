  39 /** 页面级全局：浏览器主线程里就是 `window`。 */
  40 type PageGlobal = typeof globalThis &
  41   Partial<Pick<Window, 'sessionStorage' | 'addEventListener' | 'removeEventListener'>>;
  42 
  43 /**
  44  * `sessionStorage`，拿不到时为 `null`。
  45  *
  46  * @remarks
  47  * 经 `globalThis` 而不是 `window` 取：浏览器里两者是同一个对象，而 Worker / Node 里没有 `window`。
  48  * 沙箱 iframe 与关了存储的浏览器里，**读** `sessionStorage` 这个属性本身就会抛 `SecurityError`。
  49  * 那时只是没有刷新续录，录制本身照常。
  50  */
  51 const pageStorage = (): Storage | null => {
  52   const page = globalThis as PageGlobal;
  53   if (typeof page.addEventListener !== 'function') return null;
  54   try {
  55     return page.sessionStorage ?? null;
  56   } catch {
  57     return null;
  58   }
  59 };

  95   /**
  96    * 挂页面生命周期，认领刷新暂存（有就在后台续录）。
  97    *
  98    * @remarks
  99    * 认领是同步的：键必须在本页第一次 `pagehide` 之前删掉，否则这一页的暂存会和上一页的混在同一个键里。
 100    */
 101   install(scope: LifecycleScope): void {
 102     const storage = this.#storage;
 103     if (storage === null) return;
 104     scope.acquire(() => this.#watchPage(storage), 'replay:page-lifecycle');
 105     const claim = claimReplayStash(storage, this.#stashKey);
 106     if (claim.kind === 'invalid') {
 107       this.#state.next({ kind: 'error', sessionId: claim.sessionId, error: claim.error });
 108       return;
 109     }
 110     if (claim.kind === 'stash') this.#resuming = this.#resumeOrReport(claim.stash);
 111   }

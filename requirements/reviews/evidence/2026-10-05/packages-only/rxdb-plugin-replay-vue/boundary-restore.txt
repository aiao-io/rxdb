   1 import type { RxDB } from '@aiao/rxdb';
   2 import type { WorkingTreeRestoreResult } from '@aiao/rxdb-plugin-working-tree';
   3 import { RxDBReplayError } from './errors.js';
   4 import type { ReplayRestoreRejection, ReplayRestoreResult } from './types.js';

   6 /**
   7  * 工作树插件在宿主插件索引里的名字（与 `@aiao/rxdb-plugin-working-tree` 的 `WORKING_TREE_CAPABILITY` 同值）。
   8  *
   9  * @remarks
  10  * 抄一份字面量而不是 import 那个常量：工作树是可选 peer，这里只能有类型依赖，值依赖会让没装它的应用打包失败。
  11  */
  12 export const WORKING_TREE_PLUGIN_NAME = 'workingTree';
  13 
  14 // 键取工作树的被拒原因：它新增一种原因时这里缺提示、`restoreToCommit` 的返回也对不上 `ReplayRestoreResult`，两处都编不过
  15 const RESTORE_HINTS: Readonly<Record<Extract<WorkingTreeRestoreResult, { ok: false }>['reason'], string>> = {
  16   conflict: 'The working tree changed while restoring. Try again.',
  17   dirty_working_tree: 'There are uncommitted changes. Commit or discard them before restoring.',
  18   incompatible_schema: 'This commit was written by an incompatible schema version and cannot be restored.',
  19   unreachable_target: 'This commit is not on the current branch history.'
  20 };
  21 
  22 /**
  23  * `restoreToCommit()` 被拒时给用户看的一句英文提示（`specs/005-us-909-session-replay/research.md` D8）。
  24  *
  25  * @param reason - `ReplayRestoreResult` 的 `reason`
  26  */
  27 export const replayRestoreHint = (reason: ReplayRestoreRejection): string => RESTORE_HINTS[reason];
  28 
  29 /** 这个库上装没装工作树插件（装了不等于启用，未启用由门面自己抛）。 */
  30 export const hasWorkingTree = (rxdb: RxDB): boolean => rxdb.getPlugins(WORKING_TREE_PLUGIN_NAME).length > 0;
  31 
  32 /**
  33  * 取此刻的工作树凭据，把工作树恢复到 `commitId`。
  34  *
  35  * @remarks
  36  * 凭据必须现取：回放里的标记是录制当时的事，拿那时的 HEAD 去 CAS 只会稳定地撞 `conflict`。
  37  * `status()` 与 `restore()` 之间若有人写入，CAS 照样拒成 `conflict`，由用户重试。
  38  *
  39  * @throws `RxDBReplayError('working_tree_unavailable')`：没装工作树插件；门面抛的错误原样透传
  40  */
  41 export const restoreToCommit = async (rxdb: RxDB, commitId: string): Promise<ReplayRestoreResult> => {
  42   if (!hasWorkingTree(rxdb)) {
  43     throw new RxDBReplayError(
  44       'working_tree_unavailable',
  45       'restoreToCommit() needs @aiao/rxdb-plugin-working-tree installed on this database'
  46     );
  47   }
  48   const status = await rxdb.workingTree.status();
  49   return rxdb.workingTree.restore(
  50     { commitId },
  51     {
  52       expectedBranch: { branchId: status.branchId, activationRevision: status.activationRevision },
  53       expectedHeadRevision: status.headRevision,
  54       expectedWorkingTreeRevision: status.workingTreeRevision
  55     }
  56   );
  57 };

  38 /** {@link MountReplayerSpy} 记下的一次句柄调用。 */
  39 export type ReplayerHandleCall =

  41   | { readonly method: 'play' | 'pause' | 'destroy'; readonly args: [] }

  44 /**
  45  * `mountReplayer` 的录制替身：组件测试用 `vi.mock` 把 `@aiao/rxdb-plugin-replay` 的 `mountReplayer` 指到 {@link MountReplayerSpy.mountReplayer}。
  46  *
  47  * @remarks
  48  * 不依赖 vitest，只按顺序记下挂载与句柄调用；断言由 {@link replayerParityCases} 自己做。
  49  */
  50 export class MountReplayerSpy {
  51   /** 每次 `mountReplayer(host, options)` 的参数。 */
  52   readonly mounts: { readonly host: HTMLElement; readonly options: ReplayerOptions }[] = [];
  53   /** 句柄上每次调用，按发生顺序。 */
  54   readonly calls: ReplayerHandleCall[] = [];
  55 
  56   /** 交给 `vi.mock` 的替身函数。 */
  57   readonly mountReplayer = (host: HTMLElement, options: ReplayerOptions): ReplayerHandle => {
  58     this.mounts.push({ host, options });
  59     return {
  60       update: next => this.calls.push({ method: 'update', args: [next] }),
  61       play: () => this.calls.push({ method: 'play', args: [] }),
  62       pause: () => this.calls.push({ method: 'pause', args: [] }),
  63       seek: timeMs => this.calls.push({ method: 'seek', args: [timeMs] }),
  64       destroy: () => this.calls.push({ method: 'destroy', args: [] })
  65     };
  66   };
  67 
  68   /** 清空记录；每个用例开始前调用。 */
  69   reset(): void {
  70     this.mounts.length = 0;
  71     this.calls.length = 0;
  72   }
  73 
  74   /** 某个方法的全部调用参数。 */
  75   argsOf(method: ReplayerHandleCall['method']): unknown[][] {
  76     return this.calls.filter(call => call.method === method).map(call => call.args);
  77   }
  78 }

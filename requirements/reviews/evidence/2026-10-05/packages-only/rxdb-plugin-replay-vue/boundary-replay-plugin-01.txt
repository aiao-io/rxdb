  56   async #release(runtime: ReplayRuntime): Promise<void> {
  57     if (this.#runtime === runtime) this.#runtime = null;
  58     try {
  59       await runtime.release();
  60     } finally {
  61       const state = this.#state;
  62       this.#state = new BehaviorSubject<ReplayState>(IDLE);
  63       state.complete();
  64     }
  65   }

  95 /**
  96  * 录制回放插件工厂：`rxdb.use(rxDBPluginReplay, { createRecordingDb })`。
  97  *
  98  * @remarks
  99  * 同一个库重复 `use()` 返回已装的那个实例（不比较选项）。选项非法时同步抛，插件不登记、`rxdb.replay` 不挂。
 100  *
 101  * @public
 102  */
 103 export const rxDBPluginReplay: Plugin<RxDBReplayOptions> = (db, options) => {
 104   // 问宿主的插件索引，不问 `db.replay`：门面是给使用者的，拿它当探测口会在改名时装出第二个实例
 105   const [installed] = db.getPlugins('replay');
 106   if (installed !== undefined) {
 107     if (installed instanceof RxDBPluginReplay) return installed;
 108     throw new Error('[rxdb-plugin-replay] a different plugin is already installed under the name "replay"');
 109   }
 110   const plugin = new RxDBPluginReplay(db, options);
 111   Object.defineProperty(db, 'replay', {
 112     value: plugin.manager,
 113     enumerable: false,
 114     configurable: false,
 115     writable: false
 116   });
 117   return plugin;
 118 };

   1 import {
   2   mountReplayer,
   3   type ReplayerCommitRestoreEvent,
   4   type ReplayerHandle,
   5   type ReplayerOptions
   6 } from '@aiao/rxdb-plugin-replay';
   7 import { forwardRef, useEffect, useEffectEvent, useImperativeHandle, useRef } from 'react';
   8 
   9 /** {@link Replayer} 的 props：输入与 `mountReplayer` 同名，两个回调对应核心视图的 `onTimeChange` / `onCommitRestore`。 */
  10 export interface ReplayerProps {
  11   /** 录制门面（`rxdb.replay`）。换实例会重新加载。 */
  12   readonly replay: ReplayerOptions['replay'];
  13   /** 要回放的会话。换会话会重新加载，不重建组件。 */
  14   readonly sessionId: string;
  15   /** 加载完成后落到的时刻（相对会话起点的 ms，默认 0）；之后再改等价于 `seek()`。 */
  16   readonly initialTime?: number;
  17   /** 播放 / 跳转导致回放时刻变化。 */
  18   readonly onTimeChange?: (timeMs: number) => void;
  19   /** 点 commit 标记后的恢复结果（成功、四种拒绝或抛错）。 */
  20   readonly onCommitRestore?: (event: ReplayerCommitRestoreEvent) => void;
  21 }
  22 
  23 /** 经 `ref` 拿到的命令句柄；加载完成前调用是空操作（`seek` 会记下目标时刻）。 */
  24 export type ReplayerRef = Pick<ReplayerHandle, 'play' | 'pause' | 'seek'>;
  25 
  26 type ReplayerInputs = Pick<ReplayerOptions, 'replay' | 'sessionId' | 'initialTime'>;
  27 
  28 /** 只挑出变化了的输入；都没变返回 `undefined`。 */
  29 const changedInputs = (previous: ReplayerInputs, next: ReplayerInputs): Partial<ReplayerInputs> | undefined => {
  30   const changes: { -readonly [K in keyof ReplayerInputs]?: ReplayerInputs[K] } = {};
  31   if (next.replay !== previous.replay) changes.replay = next.replay;
  32   if (next.sessionId !== previous.sessionId) changes.sessionId = next.sessionId;
  33   if (next.initialTime !== previous.initialTime) changes.initialTime = next.initialTime;
  34   return Object.keys(changes).length > 0 ? changes : undefined;
  35 };
  36 
  37 /**
  38  * 会话回放组件：在一个 `div` 里挂核心的 `mountReplayer`，输入变化走 `update()`，卸载时 `destroy()`。
  39  *
  40  * @remarks
  41  * 与 Angular `ao-replayer`、Vue `Replayer` 同一组输入 / 输出 / 命令（contracts/replayer-component.md §3），由 parity 用例保证。
  42  * `rrweb` 在核心视图里按需 `import()`，挂载这个组件之前不进包。
  43  *
  44  * @example
  45  * ```tsx
  46  * const replayer = useRef<ReplayerRef>(null);
  47  * <Replayer ref={replayer} replay={rxdb.replay} sessionId={sessionId} onCommitRestore={event => console.log(event.result)} />;
  48  * ```
  49  */
  50 export const Replayer = forwardRef<ReplayerRef, ReplayerProps>(function Replayer(
  51   { replay, sessionId, initialTime, onTimeChange, onCommitRestore },
  52   ref
  53 ) {
  54   const hostRef = useRef<HTMLDivElement>(null);
  55   const handleRef = useRef<ReplayerHandle | null>(null);
  56   const appliedRef = useRef<ReplayerInputs | null>(null);
  57 
  58   const readInputs = useEffectEvent((): ReplayerInputs => ({ replay, sessionId, initialTime }));
  59   const emitTimeChange = useEffectEvent((timeMs: number) => onTimeChange?.(timeMs));
  60   const emitCommitRestore = useEffectEvent((event: ReplayerCommitRestoreEvent) => onCommitRestore?.(event));
  61 
  62   useEffect(() => {
  63     const host = hostRef.current;
  64     if (!host) return;
  65     const inputs = readInputs();
  66     const handle = mountReplayer(host, {
  67       ...inputs,
  68       onTimeChange: timeMs => emitTimeChange(timeMs),
  69       onCommitRestore: event => emitCommitRestore(event)
  70     });
  71     handleRef.current = handle;
  72     appliedRef.current = inputs;
  73     return () => {
  74       handle.destroy();
  75       handleRef.current = null;
  76       appliedRef.current = null;
  77     };
  78   }, []);
  79 
  80   useEffect(() => {
  81     const handle = handleRef.current;
  82     const applied = appliedRef.current;
  83     if (!handle || !applied) return;
  84     const next = { replay, sessionId, initialTime };
  85     const changes = changedInputs(applied, next);
  86     if (!changes) return;
  87     appliedRef.current = next;
  88     handle.update(changes);
  89   }, [replay, sessionId, initialTime]);
  90 
  91   useImperativeHandle(
  92     ref,
  93     () => ({
  94       play: () => handleRef.current?.play(),
  95       pause: () => handleRef.current?.pause(),
  96       seek: timeMs => handleRef.current?.seek(timeMs)
  97     }),
  98     []
  99   );
 100 
 101   return <div ref={hostRef} />;
 102 });

 254 describe('update / destroy', () => {
 255   it('换 sessionId 重新加载；只改 initialTime 等价于 seek，不重新读事件', async () => {
 256     const replay = fakeReplay();
 257     const { c } = recording.offsets;
 258     mount({ replay });
 259     await ready();
 260 
 261     handle?.update({ initialTime: c + 1 });
 262     expect(probeText()).toBe('C');
 263     expect(replay.readEvents).toHaveBeenCalledTimes(1);
 264 
 265     handle?.update({ sessionId: 's2' });
 266     expect(root()?.dataset['state']).toBe('loading');
 267     await ready();
 268     expect(replay.readEvents).toHaveBeenLastCalledWith('s2');
 269     expect(host.querySelectorAll('.rxdb-replayer__stage iframe')).toHaveLength(1);
 270   });
 271 
 272   it('换会话时旧会话的加载晚到，不覆盖新会话', async () => {
 273     let releaseOld: (events: eventWithTime[]) => void = () => undefined;
 274     const readEvents = vi.fn((sessionId: string) =>
 275       sessionId === 'old' ?
 276         new Promise<eventWithTime[]>(resolve => (releaseOld = resolve))
 277       : Promise.resolve(recording.events.filter(event => event.type !== 2))
 278     );
 279     mount({ replay: fakeReplay({ readEvents }), sessionId: 'old' });
 280 
 281     handle?.update({ sessionId: 'new' });
 282     await vi.waitFor(() => expect(root()?.dataset['state']).toBe('empty'));
 283     releaseOld(recording.events);
 284     await sleep(0);
 285 
 286     expect(root()?.dataset['state']).toBe('empty');
 287   });
 288 
 289   it('destroy 清空宿主、幂等；之后的调用都是空操作，加载晚到也不再渲染', async () => {
 290     let release: (events: eventWithTime[]) => void = () => undefined;
 291     mount({ replay: fakeReplay({ readEvents: () => new Promise(resolve => (release = resolve)) }) });
 292 
 293     handle?.destroy();
 294     handle?.destroy();
 295     handle?.play();
 296     handle?.seek(10);
 297     handle?.update({ sessionId: 's2' });
 298     release(recording.events);
 299     await sleep(0);
 300 
 301     expect(host.childElementCount).toBe(0);
 302   });
 303 });

 305 describe('commit 标记', () => {
 306   const start = recording.events[0]?.timestamp ?? 0;
 307   const markers = [
 308     marker('0123456789abcdef', start + recording.offsets.b + 1),
 309     marker('fedcba9876543210', start + 61_000)
 310   ];
 311 
 312   it('每个标记一个按钮：data-commit-id，文本「id 前 8 位 · mm:ss」', async () => {
 313     mount({ replay: fakeReplay({ listCommitMarkers: async () => markers }) });
 314     await ready();
 315 
 316     const buttons = [...host.querySelectorAll<HTMLButtonElement>('ul[aria-label=Commits] > li > button')];
 317     expect(buttons.map(button => button.dataset['commitId'])).toEqual(['0123456789abcdef', 'fedcba9876543210']);
 318     expect(buttons.map(button => button.textContent)).toEqual(['01234567 · 00:00', 'fedcba98 · 01:01']);
 319   });
 320 
 321   it('点标记：跳到标记时刻并暂停，调 restoreToCommit，状态区写结果并回调宿主', async () => {
 322     const onCommitRestore = vi.fn();
 323     const replay = fakeReplay({ listCommitMarkers: async () => markers });
 324     mount({ replay, onCommitRestore });
 325     await ready();
 326     handle?.play();
 327 
 328     host.querySelector<HTMLButtonElement>('[data-commit-id="0123456789abcdef"]')?.click();
 329 
 330     expect(probeText()).toBe('B');
 331     expect(toggle()?.textContent).toBe('Play');
 332     await vi.waitFor(() => expect(statusRegion()?.textContent).toBe('Restored 3 changes'));
 333     expect(replay.restoreToCommit).toHaveBeenCalledWith('0123456789abcdef');
 334     expect(onCommitRestore).toHaveBeenCalledWith({
 335       marker: markers[0],
 336       result: { ok: true, restoredCount: 3, sessionId: 'r1', workingTreeRevision: 2 }
 337     });
 338   });
 339 
 340   it('恢复被拒：状态区写提示', async () => {
 341     const replay = fakeReplay({
 342       listCommitMarkers: async () => markers,
 343       restoreToCommit: async () => ({ ok: false, reason: 'dirty_working_tree' })
 344     });
 345     mount({ replay });
 346     await ready();
 347 
 348     host.querySelector<HTMLButtonElement>('[data-commit-id="0123456789abcdef"]')?.click();
 349 
 350     await vi.waitFor(() =>
 351       expect(statusRegion()?.textContent).toBe(
 352         'There are uncommitted changes. Commit or discard them before restoring.'
 353       )
 354     );
 355   });
 356 
 357   it('restoreToCommit 抛错：状态区写错误消息，回调收到 reason: error', async () => {
 358     const onCommitRestore = vi.fn();
 359     const error = new RxDBReplayError('working_tree_unavailable', 'no working tree here');
 360     const replay = fakeReplay({
 361       listCommitMarkers: async () => markers,
 362       restoreToCommit: async () => Promise.reject(error)
 363     });
 364     mount({ replay, onCommitRestore });
 365     await ready();
 366 
 367     host.querySelector<HTMLButtonElement>('[data-commit-id="fedcba9876543210"]')?.click();
 368 
 369     await vi.waitFor(() => expect(statusRegion()?.textContent).toBe(error.message));
 370     expect(onCommitRestore).toHaveBeenCalledWith({ marker: markers[1], result: { ok: false, reason: 'error', error } });
 371   });
 372 });

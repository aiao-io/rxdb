  18 /** 录制里每步之间的间隔（ms）：够 rrweb 把相邻两次变更分进不同时间戳。 */
  19 const STEP = 150;
  20 
  21 const sleep = (ms: number) => new Promise<void>(resolve => setTimeout(resolve, ms));
  22 
  23 interface Recording {
  24   readonly events: eventWithTime[];
  25   /** `#probe` 变成 B / C 的那条增量事件相对录像起点的时刻。 */
  26   readonly offsets: { readonly b: number; readonly c: number };
  27 }
  28 
  29 /** rrweb `IncrementalSource.Mutation` 带文本变更的那条增量事件。 */
  30 const isTextMutation = (event: eventWithTime): boolean =>
  31   event.type === 3 && 'texts' in event.data && event.data.source === 0 && event.data.texts.length > 0;
  32 
  33 /** 真 rrweb 录一段：`#probe` 文本 A → B → C（改文本节点的 `data`，rrweb 记成 `texts` 变更）。 */
  34 const recordProbe = async (): Promise<Recording> => {
  35   const probe = document.createElement('div');
  36   probe.id = 'probe';
  37   const text = document.createTextNode('A');
  38   probe.append(text);
  39   document.body.append(probe);
  40   const events: eventWithTime[] = [];
  41   const stop = record({ emit: event => events.push(event) });
  42   await sleep(STEP);
  43   text.data = 'B';
  44   await sleep(STEP);
  45   text.data = 'C';
  46   await sleep(STEP);
  47   stop?.();
  48   probe.remove();
  49   const start = events[0]?.timestamp ?? 0;
  50   const [b, c] = events.filter(isTextMutation).map(event => event.timestamp - start);
  51   if (b === undefined || c === undefined) throw new Error('录制里没找到两次文本变更');
  52   return { events, offsets: { b, c } };
  53 };
  54 
  55 const recording = await recordProbe();
  56 
  57 const marker = (commitId: string, timestamp: number): ReplayCommitMarker => ({
  58   seq: 99,
  59   timestamp,
  60   commitId,
  61   branchId: 'main'
  62 });

  64 const fakeReplay = (overrides: Partial<ReplayManager> = {}) => {
  65   const replay = {
  66     readEvents: vi.fn(async () => recording.events),
  67     listCommitMarkers: vi.fn(async (): Promise<ReplayCommitMarker[]> => []),
  68     restoreToCommit: vi.fn(async (): Promise<WorkingTreeRestoreResult> => ({
  69       ok: true,
  70       restoredCount: 3,
  71       sessionId: 'r1',
  72       workingTreeRevision: 2
  73     })),
  74     ...overrides
  75   };
  76   return replay as typeof replay & ReplayManager;
  77 };

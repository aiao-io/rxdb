   4 /**
   5  * `provideRxDB` 接受的数据库形态：实例、Promise，或返回二者之一的工厂。
   6  *
   7  * @typeParam T - 数据库类型
   8  *
   9  * @remarks
  10  * 三个框架包（Angular / React / Vue）的 provider 收的是**同一个**联合类型，
  11  * 名字也相同 —— 这是三框架对称铁律的落点之一：同一段「怎么把库交给框架」的说明
  12  * 在三端逐字成立，用户换框架时不需要重学。
  13  *
  14  * 允许 Promise 是为了「候选走动态 `import()`」这类场景（US-207 E11）：
  15  * 桌面分支不应被静态 import 打进 web bundle，而 `await import()` 只能给出 Promise。
  16  *
  17  * Vue 侧在此之上还接受 `Ref`，见 {@link RxDBInput}。
  18  */
  19 export type RxDBSource<T extends RxDB = RxDB> = T | Promise<T> | (() => T | Promise<T>);
  20 
  21 /**
  22  * `provideRxDB` 接受的数据库形态（Vue 超集）。
  23  *
  24  * @remarks
  25  * 在三端共享的 {@link RxDBSource} 之外多收两种：
  26  *
  27  * - `Ref<T | undefined>` —— 调用方自己持有的响应式引用，由调用方决定何时填上。
  28  *   这是 Vue 独有的形态，也是它先于另外两端就支持异步就绪的原因。
  29  * - `undefined` —— 尚未就绪的占位。
  30  *
  31  * 这两种都归**调用方**所有，provider 不接管它们的生命周期，见 {@link provideRxDB}。
  32  */
  33 export type RxDBInput<T extends RxDB> = RxDBSource<T> | Ref<T | undefined> | undefined;
  34 
  35 /**
  36  * injector 返回的只读数据库引用；`undefined` 表示数据库尚未就绪。
  37  */
  38 export type RxDBRef<T extends RxDB> = Readonly<Ref<T | undefined>>;

  40 /**
  41  * {@link makeRxDBDependencyInjector} 产出的一组互相隔离的 provider / injector。
  42  */
  43 export interface RxDBDependencyInjection<T extends RxDB> {
  44   /** 在当前组件上提供数据库，供其后代注入。 */
  45   provideRxDB: (db: RxDBInput<T>) => void;
  46   /** 注入当前已就绪的数据库实例；无 provider 或未就绪时返回 `undefined`。 */
  47   injectRxDB: () => T | undefined;
  48   /** 注入数据库的响应式引用；无 provider 时返回 `undefined`。 */
  49   injectRxDBRef: () => RxDBRef<T> | undefined;
  50 }
  51 
  52 /**
  53  * 注入的实际载荷：数据库引用 + 创建异常。
  54  *
  55  * @remarks
  56  * 异常单独占一格，是为了让 {@link useRxDB} 能像 Angular 的 `inject(RxDB)`、React 的
  57  * `useRxDB()` 那样把**原始创建异常**原样抛出，而不是笼统地报「还没就绪」。
  58  * 这个形状不出包，`injectRxDBRef()` 的返回类型因此保持不变。
  59  */
  60 interface RxDBSlot<T extends RxDB> {
  61   readonly db: RxDBRef<T>;
  62   readonly failure: Ref<unknown>;
  63 }

  73 /** 结构化的 thenable 判定；`await import()` 的返回值未必是原生 `Promise` 实例。 */
  74 const isThenable = <T>(value: T | PromiseLike<T>): value is PromiseLike<T> =>
  75   typeof (value as PromiseLike<T> | undefined)?.then === 'function';

  78  * source 需要异步解析吗？

  80  * @remarks
  81  * 这个判定同时定义了所有权：**要解析的才是本 provider 造出来的**，也就只有它该被断开。
  82  */
  83 const isDeferred = <T extends RxDB>(db: RxDBInput<T>): db is Promise<T> | (() => T | Promise<T>) =>
  84   typeof db === 'function' || isThenable(db);

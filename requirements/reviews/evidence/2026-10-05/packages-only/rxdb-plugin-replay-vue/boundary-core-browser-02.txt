 142 describe('跳转与播放', () => {
 143   it('seek 到某一时刻，回放 iframe 的 DOM 就是那一刻的样子；onTimeChange 收到该时刻', async () => {
 144     const onTimeChange = vi.fn();
 145     const { b, c } = recording.offsets;
 146     mount({ onTimeChange });
 147     await ready();
 148     expect(probeText()).toBe('A');
 149 
 150     const middle = Math.round((b + c) / 2);
 151     handle?.seek(middle);
 152     expect(probeText()).toBe('B');
 153     expect(Number(timeline()?.value)).toBe(middle);
 154     expect(onTimeChange).toHaveBeenLastCalledWith(middle);
 155 
 156     // c 是最后一条事件，也就是总时长：落在末尾必须看到它
 157     handle?.seek(c + 1);
 158     expect(probeText()).toBe('C');
 159     expect(Number(timeline()?.value)).toBe(c);
 160     handle?.seek(0);
 161     expect(probeText()).toBe('A');
 162   });

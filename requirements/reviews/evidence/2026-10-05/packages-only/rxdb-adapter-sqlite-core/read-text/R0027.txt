  17 |  * 递归 CTE 的失控保护上限。
  18 |  *
  19 |  * @remarks
  20 |  * 不传 `FindTreeOptions.level` 时默认不限深度，而 `WITH RECURSIVE … UNION ALL` 没有

  59 | /**
  60 |  * 构建递归成员 where 条件的字段别名映射
  61 |  *
  62 |  * @remarks
  63 |  * 递归 CTE 的递归成员以 `children` 作为表别名，此处不存在主表别名 `_`。
  64 |  * 若不映射，裸字段（如 `title`）会被 `build_rule` 生成为 `_."title"`，
  65 |  * 在递归成员中触发 “no such column: _” 运行时错误。
  66 |  * 因此将每个属性名与外键名映射到 `"children"."<columnName>"`，把裸字段绑定到递归成员表别名。
  67 |  *
  68 |  * `children.<字段>` 同样映射到递归成员自身列：这是既有的公开约定（指「递归成员自己的那一列」，
  69 |  * 而非 children 关系上的字段）。此前它靠「递归成员别名恰好也叫 children」的巧合成立，现在
  70 |  * 显式写进映射，JOIN 规划命中后会跳过，不会把它当关系路径去 JOIN 自引用表（SQLC-010）。
  71 |  */
  72 | const build_tree_field_alias_map = (metadata: EntityMetadata): Map<string, string> => {
  73 |   const fieldAliasMap = new Map<string, string>();
  74 |   const bind = (name: string, columnName: string): void => {
  75 |     const sql = `${quote_sql_identifier(TREE_RECURSIVE_ALIAS)}.${quote_sql_identifier(columnName)}`;
  76 |     fieldAliasMap.set(name, sql);
  77 |     fieldAliasMap.set(`${TREE_RECURSIVE_ALIAS}.${name}`, sql);
  78 |   };
  79 | 
  80 |   metadata.propertyMap.forEach(property => bind(property.name, property.columnName));
  81 | 
  82 |   const fkNames = metadata.foreignKeyNames ?? [];
  83 |   const fkColumnNames = metadata.foreignKeyColumnNames ?? fkNames;
  84 |   fkNames.forEach((fkName, index) => bind(fkName, fkColumnNames[index] ?? fkName));
  85 | 
  86 |   return fieldAliasMap;
  87 | };

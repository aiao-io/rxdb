 235 | /**
 236 |  * 处理 SQLite 变更事件，将 update_hook 触发的行变更转换为 RxDB 实体事件。
 237 |  *
 238 |  * 设计为 fire-and-forget：由 update_hook 同步调用，内部异步查询不阻塞调用方。
 239 |  * 失败通过 {@link RxDBAdapterSqliteBase.emitChangeError} 上报（无监听器时降级为
 240 |  * `console.error`）；这条路径没有调用方，抛出只会变成 unhandled rejection（SQLC-012）。
 241 |  * 高频场景下事件处理可能堆积，但由于 SqliteClient 内部队列保证串行执行，不会产生并发问题。
 242 |  */
 243 | export const handle_rxdb_change = <T extends EntityType>(
 244 |   adapter: RxDBAdapterSqliteBase,
 245 |   event: SqliteChangeEvent
 246 | ): Promise<void> => {
 247 |   const { tableName } = event;
 248 |   const [namespace, table_name] = get_table_name_info(tableName);
 249 |   const EntityType = adapter.rxdb.schemaManager.getEntityTypeByTableName<T>(table_name, namespace);
 250 |   if (!EntityType) return Promise.resolve();
 251 |   const repository = adapter.getRepository<T, SqliteRepository<T>>(EntityType);
 252 |   const metadata = getEntityMetadata(EntityType);
 253 |   const rxdb_change_meta = getEntityMetadata(RxDBChange);
 254 | 
 255 |   const { rowIds } = event;
 256 |   return repository
 257 |     .findByRowIds(rowIds, event.type === SQLiteChangeType.SQLITE_UPDATE)
 258 |     .then(async result => {
 259 |       try {
 260 |         if (metadata === rxdb_change_meta) {
 261 |           await handle_rxdb_change_table(adapter, result, event, metadata);
 262 |         } else {
 263 |           handle_system_table(adapter, result, event, metadata);
 264 |         }
 265 |       } catch (error) {
 266 |         adapter.emitChangeError({ error, tableName, rowIds, phase: 'process' });
 267 |       }
 268 |     })
 269 |     .catch(error => {
 270 |       // 断开期间被取消的查询不是失败：没有变更丢失，只是这个 adapter 不再服务
 271 |       if (error instanceof RxDBAdapterSqliteError && error.code === 'adapter_disconnected') return;
 272 |       adapter.emitChangeError({ error, tableName, rowIds, phase: 'query' });
 273 |     });
 274 | };

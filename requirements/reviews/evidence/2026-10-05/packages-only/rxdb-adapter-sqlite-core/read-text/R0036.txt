  21 | export const generate_upsert_clause = (primaryKeyColumn: string, columns: readonly string[]): string => {
  22 |   const conflictTarget = quote_sql_identifier(primaryKeyColumn);
  23 |   const updateColumns = columns.filter(column => column !== primaryKeyColumn);
  24 |   if (updateColumns.length === 0) return ` ON CONFLICT (${conflictTarget}) DO NOTHING`;
  25 |   const assignments = updateColumns
  26 |     .map(column => `${quote_sql_identifier(column)} = excluded.${quote_sql_identifier(column)}`)
  27 |     .join(',');
  28 |   return ` ON CONFLICT (${conflictTarget}) DO UPDATE SET ${assignments}`;
  29 | };

  43 |   // 两个键在 transformEntityValueToSql 里落到同一个物理列上、后写的赢 —— 表现是

  47 |   const columnOf = (propertyName: string): string | undefined => metadata.propertyMap.get(propertyName)?.columnName;
  48 | 
  49 |   if (context?.userId) {
  50 |     const createdByColumn = columnOf('createdBy');
  51 |     const updatedByColumn = columnOf('updatedBy');
  52 |     if (createdByColumn !== undefined) columns[createdByColumn] = context.userId;
  53 |     if (updatedByColumn !== undefined) columns[updatedByColumn] = context.userId;
  54 |   }
  55 |

  58 |   if (createdAtColumn !== undefined && columns[createdAtColumn] === undefined) {

  60 |   }
  61 |   const updatedAtColumn = columnOf('updatedAt');
  62 |   if (updatedAtColumn !== undefined && columns[updatedAtColumn] === undefined) {
  63 |     columns[updatedAtColumn] = context?.updatedAt ?? now;
  64 |   }

  67 |   const setColumns = Object.keys(needSaveData);

  69 |   const params = Object.values(needSaveData) as SQLiteCompatibleType[];

  71 |   let sql = `INSERT INTO ${quote_sql_identifier(tableName)} (${setColumns.map(quote_sql_identifier).join(',')}) VALUES (${setPlaceholders})`;

  80 |   }

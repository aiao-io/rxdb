 203 | /**
 204 |  * 将 SQLite 类型值转换为 JS 类型
 205 |  * @param value SQLite 类型的值
 206 |  * @param property 实体属性元数据
 207 |  * @returns JS 类型的值
 208 |  */
 209 | export const transformValueSqliteToJs = (
 210 |   value: unknown,
 211 |   property: EntityPropertyMetadata | KeyValuePropertyMetadata
 212 | ) => {
 213 |   if (value === null && property.nullable) return null;
 214 |   if (value === void 0) return void 0;
 215 |   switch (property.type) {
 216 |     case PropertyType.bigint:
 217 |       if (typeof value === 'bigint') return assertSigned64BitBigInt(value, property.name);
 218 |       if (typeof value === 'number' && Number.isSafeInteger(value)) return BigInt(value);
 219 |       throw new TypeError(`${property.name} did not read as a lossless signed 64-bit bigint`);
 220 |     case PropertyType.binary:
 221 |       if (value instanceof Uint8Array) return new Uint8Array(value);
 222 |       if (Array.isArray(value) && value.every(item => Number.isInteger(item) && item >= 0 && item <= 255)) {
 223 |         return new Uint8Array(value);
 224 |       }
 225 |       throw new TypeError(`${property.name} did not read as binary data`);
 226 |     case PropertyType.boolean:
 227 |       return Boolean(value);
 228 |     case PropertyType.date: {
 229 |       if (value === '') return null; // 空字符串不应产生 Invalid Date
 230 |       if (value instanceof Date) return isNaN(value.getTime()) ? null : value;
 231 |       if (typeof value === 'string') {
 232 |         const d = new Date(value);
 233 |         return isNaN(d.getTime()) ? null : d;
 234 |       }
 235 |       return value;
 236 |     }
 237 |     case PropertyType.keyValue: {
 238 |       if (!value) return null;
 239 |       const parsed: unknown = typeof value === 'string' ? JSON.parse(value) : value;
 240 |       if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) return parsed;
 241 |       const converted: Record<string, unknown> = { ...parsed };
 242 |       property.properties?.forEach(nestedProp => {
 243 |         if (nestedProp.name in converted) {
 244 |           converted[nestedProp.name] = transformValueSqliteToJs(converted[nestedProp.name], nestedProp);
 245 |         }
 246 |       });
 247 |       return converted;
 248 |     }
 249 |     case PropertyType.stringArray:
 250 |     case PropertyType.numberArray: {
 251 |       if (Array.isArray(value)) return value;
 252 |       if (typeof value !== 'string') return value ? value : null;
 253 |       const parsed: unknown = JSON.parse(value);
 254 |       return parsed;
 255 |     }
 256 |     case PropertyType.json:
 257 |       if (value === null) return null;
 258 |       if (typeof value !== 'string') return value;
 259 |       if (value === '') return null;
 260 |       try {
 261 |         const parsed: unknown = JSON.parse(value);
 262 |         return parsed;
 263 |       } catch {
 264 |         return value;
 265 |       }
 266 |     default:
 267 |       return value;
 268 |   }
 269 | };

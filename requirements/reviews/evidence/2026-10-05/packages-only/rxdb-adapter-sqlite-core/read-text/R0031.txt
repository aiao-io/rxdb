  51 | const toSQLiteCompatibleValue = (value: unknown, fieldName: string): SQLiteCompatibleType | undefined => {
  52 |   if (value === undefined || value === null) return value;
  53 |   if (typeof value === 'string' || typeof value === 'number' || typeof value === 'bigint') return value;

 105 | export const chunkBySqliteBindLimit = <T>(items: readonly T[], variablesPerItem = 1): T[][] => {
 106 |   if (items.length === 0) {
 107 |     return [];
 108 |   }
 109 | 
 110 |   const chunkSize = getSqliteBindChunkSize(variablesPerItem);
 111 |   if (items.length <= chunkSize) {
 112 |     return [items.slice()];
 113 |   }
 114 | 
 115 |   const chunks: T[][] = [];
 116 |   for (let i = 0; i < items.length; i += chunkSize) {
 117 |     chunks.push(items.slice(i, i + chunkSize));
 118 |   }
 119 |   return chunks;
 120 | };

 122 | /**
 123 |  * 将 RxDB 属性类型转换为 SQLite 数据类型
 124 |  */
 125 | export const rxDBColumnTypeToSqliteType = (property: {
 126 |   readonly type: string;
 127 |   readonly encrypted?: boolean;
 128 | }): SqliteDataType => {
 129 |   let type: SqliteDataType;
 130 |   switch (property.type) {
 131 |     case PropertyType.uuid:
 132 |     case PropertyType.string:
 133 |     case PropertyType.enum:
 134 |     case PropertyType.json:
 135 |     case PropertyType.keyValue:
 136 |     case PropertyType.stringArray:
 137 |     case PropertyType.numberArray:
 138 |     case PropertyType.date:
 139 |       type = 'TEXT';
 140 |       break;
 141 |     case PropertyType.number:
 142 |       type = 'REAL';
 143 |       break;
 144 |     case PropertyType.integer:
 145 |     case PropertyType.boolean:
 146 |     case PropertyType.bigint:
 147 |       type = 'INTEGER';
 148 |       break;
 149 |     case PropertyType.binary:
 150 |       type = 'BLOB';
 151 |       break;
 152 |     default:
 153 |       throw new RxDBAdapterSqliteError(`Unsupported property type: ${String(property.type)}`);
 154 |   }
 155 |   return property.encrypted === true ? 'TEXT' : type;
 156 | };

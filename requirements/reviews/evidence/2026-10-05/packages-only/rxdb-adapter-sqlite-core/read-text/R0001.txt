  17 | /**
  18 |  * 把一个适配器包装成「所有 `query()` 都落在指定事务里」的门面。
  19 |  *
  20 |  * @remarks
  21 |  * 这是 C2 的关键简化：仓库与内部 helper（`rxdb_adapter_mutations` / `execute_switch_actions` …）
  22 |  * 一律通过 `adapter.query()` 访问数据库。给它们一个 `query` 被改写过的门面，就等于把整段调用链
  23 |  * 拉进事务，**不需要给每个 helper 逐个加 executor 形参**。
  24 |  *
  25 |  * 绑定规则（两条都必须遵守，否则运行时会炸）：
  26 |  * - 除 `query` / `getRepository` 外的所有成员一律取自并绑定到**真实适配器**。
  27 |  *   适配器方法内部大量访问 `#private` 字段，而 `#` 字段访问不走 Proxy 的 `get` 陷阱 ——
  28 |  *   绑到门面上会直接 `TypeError: Cannot read private member`。
  29 |  * - `getRepository` 必须返回**新建**且绑定到门面的仓库，且**绝不写入** `repository_cache`：
  30 |  *   写进去的话事务结束后缓存里会留着一个指向已提交/已回滚连接的仓库，
  31 |  *   之后的**外部**写会打到那个死连接上。
  32 |  */
  33 | const createExecutorAdapterFacade = (
  34 |   adapter: RxDBAdapterSqliteBase,
  35 |   executor: SqliteTransactionExecutor
  36 | ): RxDBAdapterSqliteBase =>
  37 |   new Proxy(adapter, {
  38 |     get(target, property) {
  39 |       if (property === 'query') {
  40 |         return (sql: string, bindings?: SQLiteCompatibleType[]): Promise<SqliteResult> =>
  41 |           executor.execute(sql, bindings);
  42 |       }
  43 |       if (property === 'writeQuery') {
  44 |         return (sql: string, bindings?: SQLiteCompatibleType[]): Promise<SqliteResult> =>
  45 |           executor.execute(sql, bindings);
  46 |       }
  47 |       if (property === 'getRepository') {
  48 |         return (EntityType: EntityType): unknown => executor.getRepository(EntityType);
  49 |       }
  50 |       if (property === 'runInTransaction') {
  51 |         // 已经在本事务里了：复用，绝不新开也绝不入队，否则内部 helper 会在自己持槽时再入队而永久挂起。
  52 |         return (fun: (executor: TransactionExecutor) => Promise<unknown>): Promise<unknown> => executor.run(fun);
  53 |       }
  54 |       if (property === 'mergeChanges') {
  55 |         // `this` 必须绑到门面：mergeChanges 把收到的 adapter 一路传给内部 helper，
  56 |         // 绑到真实适配器会让它们的 query 走队列（翻转后即自锁）。
  57 |         const mergeChanges = Reflect.get(target, property, target) as (...args: unknown[]) => unknown;
  58 |         return (...args: unknown[]): unknown => mergeChanges.apply(executor.adapter, args);
  59 |       }
  60 |       // receiver 一律**丢弃**，取值与绑定都用真实适配器：getter 里访问 `#private` 字段不走
  61 |       // Proxy，传门面进去就是 `TypeError: Cannot read private member`（见上面的绑定规则）。
  62 |       // 此前写成 `receiver === undefined ? target : target` —— 两支相同，却把这行读成
  63 |       // 「按 receiver 分派」。真正被特判的只有上面那四个成员名，其余一概直达真实适配器；
  64 |       // 尤其 `runInTransaction` 有特判而 `transaction` 没有（后者必须重新入队），
  65 |       // 误以为这里还有一层分派的人会算错哪些调用会撞上并发度为 1 的队列。
  66 |       const value: unknown = Reflect.get(target, property, target);
  67 |       return typeof value === 'function' ? (value as (...args: unknown[]) => unknown).bind(target) : value;
  68 |     }
  69 |   });

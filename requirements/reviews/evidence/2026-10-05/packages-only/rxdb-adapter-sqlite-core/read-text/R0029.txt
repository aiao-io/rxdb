  41 | /**
  42 |  * 校验 Comlink 传输选项的组合合法性：worker / workerInstance / sharedWorker 必须成对出现。
  43 |  */
  44 | export function validateComlinkTransportOptions(options: CreateSqliteClientOptions): void {
  45 |   const hasWorkerFlag = options.worker === true;
  46 |   const hasWorkerInstance = options.workerInstance !== undefined;
  47 |   const hasSharedWorkerFlag = options.sharedWorker === true;
  48 |   const hasSharedWorkerInstance = options.sharedWorkerInstance !== undefined;
  49 | 
  50 |   if (hasWorkerFlag && !hasWorkerInstance) {
  51 |     throw new RxDBAdapterSqliteError('`worker` and `workerInstance` must be provided together');
  52 |   }
  53 | 
  54 |   if (options.worker === false && hasWorkerInstance) {
  55 |     throw new RxDBAdapterSqliteError('`worker` cannot be false when `workerInstance` is provided');
  56 |   }
  57 | 
  58 |   if (hasSharedWorkerFlag && !hasSharedWorkerInstance) {
  59 |     throw new RxDBAdapterSqliteError('`sharedWorker` and `sharedWorkerInstance` must be provided together');
  60 |   }
  61 | 
  62 |   if (options.sharedWorker === false && hasSharedWorkerInstance) {
  63 |     throw new RxDBAdapterSqliteError('`sharedWorker` cannot be false when `sharedWorkerInstance` is provided');
  64 |   }
  65 | 
  66 |   if ((hasWorkerFlag || hasWorkerInstance) && (hasSharedWorkerFlag || hasSharedWorkerInstance)) {
  67 |     throw new RxDBAdapterSqliteError('Worker and SharedWorker transports are mutually exclusive');
  68 |   }
  69 | 
  70 |   if (
  71 |     options.workerOwnership !== undefined &&
  72 |     options.workerOwnership !== 'caller' &&
  73 |     options.workerOwnership !== 'client'
  74 |   ) {
  75 |     throw new RxDBAdapterSqliteError('`workerOwnership` must be either `caller` or `client`');
  76 |   }
  77 | 
  78 |   if (options.workerOwnership !== undefined && !hasWorkerInstance && !hasSharedWorkerInstance) {
  79 |     throw new RxDBAdapterSqliteError('`workerOwnership` requires a Worker or SharedWorker transport');
  80 |   }
  81 | }

  83 | /**
  84 |  * 断言即将跨 worker 传递的 load options 可被结构化克隆。
  85 |  *
  86 |  * Comlink 用 `postMessage` 传参，函数不在结构化克隆算法的支持范围内：把
  87 |  * `locateFile` / `print` / `printErr` 这类回调塞进 `client.init(dbName, loadOptions)`
  88 |  * 会在 worker 边界抛 `DataCloneError`，而报错栈里看不出是哪个选项造成的。
  89 |  *
  90 |  * 因此在跨线程前显式拦截并点名选项 —— 既不静默丢弃回调（会让 WASM 定位悄悄走错路径），
  91 |  * 也不降级回主线程（会伪装出线程隔离的假象）。
  92 |  *
  93 |  * @param loadOptions - 将被传给 `client.init` 的运行期选项
  94 |  * @param options - transport 配置，用于判断是否真的跨 worker
  95 |  * @throws worker / sharedWorker 模式下存在函数型选项时抛 {@link RxDBAdapterSqliteError}
  96 |  */
  97 | export function assertLoadOptionsTransferable(
  98 |   loadOptions: Readonly<Record<string, unknown>>,
  99 |   options: CreateSqliteClientOptions
 100 | ): void {
 101 |   if (options.workerInstance === undefined && options.sharedWorkerInstance === undefined) return;
 102 | 
 103 |   const functionKeys = Object.keys(loadOptions).filter(key => typeof loadOptions[key] === 'function');
 104 |   if (functionKeys.length === 0) return;
 105 | 
 106 |   throw new RxDBAdapterSqliteError(
 107 |     `Cannot transfer function options to a worker: ${functionKeys.join(', ')}. ` +
 108 |       'Functions are not structured-cloneable. Use `wasmPath` (a string URL) to locate WASM assets, ' +
 109 |       'and keep `print` / `printErr` on the main thread.'
 110 |   );
 111 | }

 113 | /**
 114 |  * 提供 Worker/SharedWorker 选项时使用 Comlink 包装客户端，否则返回直接客户端。
 115 |  *
 116 |  * @remarks
 117 |  * 返回类型是联合而非裸 `T`：Comlink 给回的是 {@link Remote}，
 118 |  * 它把每个方法的返回值 Promise 化，并且**不带** `T` 的私有字段与同步返回签名。
 119 |  * 早先用 `as unknown as T` 把远端代理断言成本地客户端类型，
 120 |  * 于是 `beginTransactionSql(): string` 这类同步签名在 worker 模式下是假的
 121 |  * —— 调用方按类型不 await，拿到的是 Promise 而不是 SQL 文本。
 122 |  * 调用方应把结果当作 {@link SqliteClientLike}（其签名已按「远端会 Promise 化」放宽），
 123 |  * 而不是具体的客户端实现类。
 124 |  *
 125 |  * @param directClient - 主线程客户端实例，未配置 transport 时原样返回
 126 |  * @param options - transport 配置
 127 |  * @returns 主线程客户端或其 Comlink 远端代理
 128 |  */
 129 | export function wrapWithComlink<T extends SqliteClientLike>(
 130 |   directClient: T,
 131 |   options: CreateSqliteClientOptions
 132 | ): T | Remote<T> {
 133 |   validateComlinkTransportOptions(options);
 134 | 
 135 |   if (options.workerInstance) {
 136 |     return wrap<T>(options.workerInstance);
 137 |   }
 138 |   if (options.sharedWorkerInstance) {
 139 |     return wrap<T>(options.sharedWorkerInstance.port);
 140 |   }
 141 |   return directClient;
 142 | }

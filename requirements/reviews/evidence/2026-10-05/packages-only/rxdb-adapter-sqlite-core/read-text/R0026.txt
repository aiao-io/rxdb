  14 |   adapter: RxDBAdapterSqliteBase,

  16 |   options: CountOptions<EntityType, object>

  19 |     throw new RxDBAdapterSqliteError('Invalid count query rule group');
  20 |   }
  21 |   const whereOptions = options.where;
  22 |   if (options.groupBy) {
  23 |     throw new RxDBAdapterSqliteError('groupBy not supported yet');

  26 |     adapter.rxdb.schemaManager.getEntityMetadata(name, namespace ?? metadata.namespace)

  29 |   const tableName = get_table_name_by_metadata(metadata);
  30 |   const params: SQLiteCompatibleType[] = [];
  31 |   const where = buildRuleGroup(whereOptions, fieldAliasMap, metadata, adapter, params);
  32 |   // 有 JOIN 时必须 COUNT(DISTINCT ...)：一个父实体匹配 N 条关联行会被 JOIN 展开成 N 行，
  33 |   // 裸 COUNT 会把它计 N 次。find_sql 在 hasJoin 时已经加 DISTINCT（见 generate_sql），
  34 |   // count 不同口径就会与 find(...).length 对不上（SQLC-009）
  35 |   const countExpr = joinSQL ? `COUNT(DISTINCT ${MAIN_TABLE_ALIAS}.rowid)` : `COUNT(${MAIN_TABLE_ALIAS}.rowid)`;
  36 |   const parts = [`SELECT ${countExpr} AS count FROM ${quote_sql_identifier(tableName)} ${MAIN_TABLE_ALIAS}`];
  37 |   if (joinSQL) parts.push(joinSQL);
  38 |   if (where) parts.push(` WHERE ${where}`);
  39 |   const sql = parts.join('');

  32 | /**
  33 |  * 把变更行里的加密列解回明文。
  34 |  *
  35 |  * @remarks
  36 |  * 触发器写进 change 行的是物理列的原值，而加密列在库里是 TEXT 信封 —— 不解 envelope
  37 |  * 就直接派发，等于把密文交给 QueryManager/UI（SQLC-011）。
  38 |  *
  39 |  * 解密结果只存在于返回值里，**不回写变更行**：change 表自身的元事件仍要携带原始信封。
  40 |  *
  41 |  * `unenvelopePlaintextPatches` 只处理字符串值，因此 SQLC-011 修复前写下的加密 boolean
  42 |  * 历史行（触发器把信封压成了数字 0/1）会原样透传，而不是在解密处炸掉整条变更流。
  43 |  */
  44 | const decrypt_change_patches = async (adapter: RxDBAdapterSqliteBase, change: RxDBChange): Promise<DecryptedChange> => {
  45 |   const patch = (change.patch ?? null) as Record<string, unknown> | null;
  46 |   const inversePatch = (change.inversePatch ?? null) as Record<string, unknown> | null;
  47 |   const { keyring } = adapter.encryptionContext;
  48 |   if (!keyring) return { change, patch, inversePatch };
  49 |   const entity = adapter.rxdb.schemaManager.getEntityMetadata(change.entity, change.namespace);
  50 |   if (!entity?.encryptedPropertyMap?.size) return { change, patch, inversePatch };
  51 | 
  52 |   const primaryKeyString = change.entityId;
  53 |   const unenvelope = (target: Record<string, unknown>): Promise<Record<string, unknown>> =>
  54 |     unenvelopePlaintextPatches({ entity, primaryKeyString, patch: target, keyring });
  55 |   return {
  56 |     change,
  57 |     patch: patch === null ? null : await unenvelope(patch),
  58 |     inversePatch: inversePatch === null ? null : await unenvelope(inversePatch)
  59 |   };
  60 | };

  62 | const emit_change_event = async (adapter: RxDBAdapterSqliteBase, eventData: RxDBChange[]): Promise<void> => {

  64 |   const group = new Map<RxDBChange['type'], DecryptedChange[]>();
  65 |   for (const item of decrypted) {
  66 |     const items = group.get(item.change.type);
  67 |     if (items) items.push(item);
  68 |     else group.set(item.change.type, [item]);
  69 |   }
  70 | 
  71 |   const inserts = group.get('INSERT');
  72 |   if (inserts?.length) {
  73 |     const events = inserts.map(({ change, patch }): RxDBEntityLocalCreatedEventData => ({
  74 |       namespace: change.namespace!,
  75 |       entity: change.entity,
  76 |       type: change.type,
  77 |       id: change.entityId,
  78 |       patch: patch as unknown as EntityFull<EntityType>,
  79 |       inversePatch: null,
  80 |       recordAt: change.createdAt
  81 |     }));

  83 |   }
  84 | 
  85 |   const updates = group.get('UPDATE');
  86 |   if (updates?.length) {

  88 |       namespace: change.namespace!,
  89 |       entity: change.entity,

  97 |   }
  98 |

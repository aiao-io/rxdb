 599 |   public async disconnect() {
 600 |     this.#lifecycle_state = 'closing';
 601 |     this.#is_disconnected = true;
 602 |     // client.disconnect() 失败也必须清掉缓存：不清的话这个（可能已部分拆卸的）死实例会一直
 603 |     // 留在 #cached_client 里，重连时 #client() 直接复用它而不是走工厂重建，
 604 |     // 后续每次 disconnect 重试也会对同一个实例反复调用（SQLC-020）
 605 |     try {
 606 |       await this.#client_promise?.catch(() => undefined);
 607 |       await this.#waitForChangeTasks();
 608 |       await this.#queue.waitForAll();
 609 |       if (this.#cached_client) {
 610 |         await this.#cached_client.disconnect();
 611 |       }
 612 |     } finally {
 613 |       await this.#releaseAdoptedClient();
 614 |       await this.#releaseStorageLock();
 615 |       // client 是 Comlink 远端代理时必须显式释放根代理的 MessagePort：
 616 |       // 不释放则每轮断开/重连都留下一个活端口和 Worker 侧引用（SQLC-041）。
 617 |       // 排在 client.disconnect() 之后 —— 先释放代理，后续 RPC 会直接 reject。
 618 |       releaseComlinkProxy(this.#cached_client);
 619 |       this.#cached_client = undefined;
 620 |       this.#client_promise = undefined;
 621 |       this.#listeners_registered = false;
 622 |       this.#lifecycle_state = 'closed';
 623 |     }
 624 |   }

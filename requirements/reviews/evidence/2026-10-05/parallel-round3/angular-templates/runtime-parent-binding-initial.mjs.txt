function makeTreeCore() {
  const { TemplateNode } = components;
  database = new RxDB({ dbName: 'r3-angular-template-tree', entities: [TemplateNode], sync: { type: SyncType.None, local: { adapter: 'controlled-boundary' } }, multiInstance: false });
  database.use(rxDBPluginTree);
  database.init();
  const pending = [];
  const calls = { findDescendants: [], findAncestors: [], countDescendants: [], countAncestors: [] };
  const backend = {
    findDescendants(options) {
      calls.findDescendants.push(options);
      const deferred = Promise.withResolvers();
      pending.push({ options, ...deferred });
      return deferred.promise;
    },
    async findAncestors(options) { calls.findAncestors.push(options); return []; },
    async countDescendants(options) { calls.countDescendants.push(options); return 0; },
    async countAncestors(options) { calls.countAncestors.push(options); return 0; }
  };
  let subscriptions = 0;
  const primary = new Observable(subscriber => {
    subscriptions += 1;
    subscriber.next(backend);
    return () => { subscriptions -= 1; };
  });
  const repository = database.entityManager.getRepository(TemplateNode, TreeRepository);
  Object.defineProperty(repository, 'primary$', { value: primary });
  return { pending, calls, subscriptions: () => subscriptions, row: (id, label) => database.entityManager.createEntityRef(TemplateNode, { id, label, parentId: null }) };
}

function makeSource() {
  return {
    search(query, options) {
      const record = { query, options, subscriptions: 0, destroyed: 0, released: 0 };
      const core = createSearchHandle({
        initialQuery: query, debounceMs: options?.debounce ?? 0, refreshAuditMs: 0,
        performSearch: async queryText => {
          searchExecutions.push(queryText);
          if (queryText === 'failure') throw new SearchExecutionError('R3 controlled executor failure');
          return { results: [{ entity: 'Article', collection: 'article', id: queryText, rank: -1, matchedField: 'title', snippet: queryText }], hasMore: false };
        },
        subscribeDataChanges: () => () => { record.released += 1; }
      });
      const observe = stream => new Observable(subscriber => {
        record.subscriptions += 1;
        const subscription = stream.subscribe(subscriber);
        return () => { record.subscriptions -= 1; subscription.unsubscribe(); };
      });
      searchRecords.push(record);
      return { ...core, results$: observe(core.results$), state$: observe(core.state$), error$: observe(core.error$), hasMore$: observe(core.hasMore$), destroy() { record.destroyed += 1; core.destroy(); } };
    }
  };
}

async function settle() {
  fixture.detectChanges();
  await Promise.resolve();
  TestBed.tick();
  await fixture.whenStable();
  fixture.detectChanges();
}

async function waitUntil(predicate) {
  const deadline = Date.now() + 1800;
  while (!predicate()) {
    if (Date.now() >= deadline) throw new Error('R3 controlled executor did not settle');
    await pause(3);
    await settle();
  }
}

async function treeAot() {
  treeCore = makeTreeCore();
  TestBed.configureTestingModule({ imports: [components.TreeParent], providers: [provideZonelessChangeDetection(), provideRxDB(database)] });
  fixture = TestBed.createComponent(components.TreeParent);
  await settle();
  check('AOT parent binds required rootId=0 without setInput', () => assert.equal(fixture.nativeElement.querySelector('[data-root]').textContent, '0'));
  check('AOT genuine signal input metadata includes rootId', () => assert.equal(metadata.TreeChild.rootId[0], 'rootId'));
  for (const [method, calls] of Object.entries(treeCore.calls)) {
    check(`${method} receives numeric zero and level one`, () => assert.deepEqual(calls, [{ entityId: 0, level: 1 }]));
  }
  check('four lazy wrapper queries have four live primary subscriptions', () => assert.equal(treeCore.subscriptions(), 4));
  check('parent bound child renders loading', () => assert.ok(fixture.nativeElement.querySelector('[data-loading]')));
  const old = treeCore.pending[0];
  fixture.componentInstance.rootId.set(1);
  fixture.componentInstance.rootId.set(2);
  await settle();
  const current = treeCore.pending.at(-1);
  check('parent signal fast update reaches current rootId=2', () => assert.deepEqual(current.options, { entityId: 2, level: 1 }));
  current.resolve([treeCore.row(2, 'current')]);
  await settle();
  old.resolve([treeCore.row(0, 'old')]);
  await settle();
  check('late old query cannot overwrite current DOM', () => assert.equal(fixture.nativeElement.querySelector('[data-node]').textContent, 'current'));
  check('numeric output is received by parent template', () => { fixture.nativeElement.querySelector('button').click(); assert.equal(fixture.componentInstance.selected(), 2); });
  fixture.componentInstance.rootId.set(3);
  await settle();
  treeCore.pending.at(-1).resolve([]);
  await settle();
  check('empty response is rendered by bound child', () => assert.ok(fixture.nativeElement.querySelector('[data-empty]')));
  fixture.componentInstance.rootId.set(4);
  await settle();
  treeCore.pending.at(-1).reject(new Error('tree-template-error'));
  await settle();
  check('query failure is rendered as alert', () => assert.equal(fixture.nativeElement.querySelector('[role=alert]').textContent, 'tree-template-error'));
  fixture.destroy();
  check('destroy releases all four primary subscriptions', () => assert.equal(treeCore.subscriptions(), 0));
}

async function searchLifecycle() {
  const source = makeSource();
  TestBed.configureTestingModule({ imports: [components.LifecycleSearchParent], providers: [provideZonelessChangeDetection(), { provide: components.SEARCH_SOURCE, useValue: source }] });
  fixture = TestBed.createComponent(components.LifecycleSearchParent);
  await settle();
  await waitUntil(() => fixture.nativeElement.querySelector('[data-result]')?.textContent === 'seed');
  check('required source and options are available at ngOnInit', () => assert.equal(searchRecords.length, 1));
  check('real SearchHandle receives bound seed/options', () => assert.deepEqual({ query: searchRecords[0].query, options: searchRecords[0].options }, { query: 'seed', options: { debounce: 0, initialQuery: 'seed' } }));
  check('real handle has four observable subscriptions', () => assert.equal(searchRecords[0].subscriptions, 4));
  const element = fixture.nativeElement.querySelector('input');
  element.value = 'typed';
  element.dispatchEvent(new window.Event('input', { bubbles: true }));
  await settle();
  await waitUntil(() => fixture.nativeElement.querySelector('[data-result]')?.textContent === 'typed');
  check('DOM Event narrows to string and updates real query', () => assert.ok(searchExecutions.includes('typed')));
  fixture.componentInstance.options.set({ debounce: 0, pageSize: 2 });
  await settle();
  await waitUntil(() => searchRecords.length === 2 && fixture.nativeElement.querySelector('[data-result]')?.textContent === 'typed');
  check('bound required options change rebuilds current handle', () => assert.deepEqual(searchRecords[1].options, { debounce: 0, pageSize: 2 }));
  check('rebuild retains current query instead of stale seed', () => assert.equal(searchRecords[1].query, 'typed'));
  check('rebuild releases old subscriptions and destroy once', () => assert.deepEqual({ subscriptions: searchRecords[0].subscriptions, destroyed: searchRecords[0].destroyed }, { subscriptions: 0, destroyed: 1 }));
  element.value = 'failure';
  element.dispatchEvent(new window.Event('input', { bubbles: true }));
  await settle();
  await waitUntil(() => fixture.nativeElement.querySelector('[role=alert]'));
  check('controlled real handle execution error reaches alert', () => assert.match(fixture.nativeElement.querySelector('[role=alert]').textContent, /R3 controlled executor failure/));
  fixture.nativeElement.querySelector('button').click();
  await settle();
  check('clear removes results and returns idle', () => { assert.equal(fixture.nativeElement.querySelector('[data-result]'), null); assert.equal(fixture.nativeElement.querySelector('[data-state]').textContent, 'idle'); });
  fixture.destroy();
  check('destroy releases every real handle stream and callback', () => assert.ok(searchRecords.every(record => record.subscriptions === 0 && record.destroyed === 1 && record.released === 1)));
}
